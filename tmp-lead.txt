// Google Apps Script: receives a site form and sends the lead to VK from a community.
// Script properties: VK_TOKEN, VK_PEER_ID, ALLOWED_SITES (comma-separated site hostnames)
const LABELS = { site: "Сайт", name: "Имя", phone: "Телефон", details: "Заявка", comment: "Комментарий" };
const MAX_LEADS_PER_HOUR = 30;
const clean = v => String(v ?? "").replace(/\s+/g, " ").trim().slice(0, 500);

// Apps Script can't set a status code, so the result is only in the body
const reply = body => ContentService.createTextOutput(JSON.stringify(body)).setMimeType(ContentService.MimeType.JSON);

function doPost(e) {
  const env = PropertiesService.getScriptProperties().getProperties();

  let form;
  try { form = JSON.parse(e.postData.contents); } catch (err) { return reply({ ok: false, error: "bad_json" }); }

  // Hidden field: real visitors leave it empty, simple bots fill it in
  if (form.website) return reply({ ok: true });

  // Apps Script doesn't expose the Origin header: this only filters out careless senders
  const allowed = (env.ALLOWED_SITES || "").split(",").map(s => s.trim()).filter(Boolean);
  if (!allowed.includes(clean(form.site))) return reply({ ok: false, error: "forbidden" });

  const digits = String(form.phone ?? "").replace(/\D/g, "");
  if (clean(form.name).length < 2 || digits.length !== 11) return reply({ ok: false, error: "bad_contact" });

  // The URL is public and has no origin check, so cap how many leads can reach VK
  const cache = CacheService.getScriptCache();
  const hourKey = "leads_" + Math.floor(Date.now() / 3600000);
  const sentThisHour = Number(cache.get(hourKey) || 0);
  if (sentThisHour >= MAX_LEADS_PER_HOUR) return reply({ ok: false, error: "rate_limited" });

  const lines = ["Новая заявка с сайта"];
  for (const [key, label] of Object.entries(LABELS)) if (clean(form[key])) lines.push(`${label}: ${clean(form[key])}`);

  try {
    const res = UrlFetchApp.fetch("https://api.vk.com/method/messages.send", {
      method: "post",
      muteHttpExceptions: true,
      payload: {
        access_token: env.VK_TOKEN,
        v: "5.199",
        peer_id: env.VK_PEER_ID,
        random_id: String(Math.floor(Math.random() * 2147483647)),
        message: lines.join("\n"),
      },
    });
    const data = JSON.parse(res.getContentText());
    if (data.error) {
      // Log only the VK code and text: the request itself contains the token
      console.log(`VK error ${data.error.error_code}: ${data.error.error_msg}`);
      return reply({ ok: false, error: "send_failed" });
    }
  } catch (err) {
    console.log(`send failed: ${err.name}`);
    return reply({ ok: false, error: "send_failed" });
  }
  cache.put(hourKey, String(sentThisHour + 1), 3600);
  return reply({ ok: true });
}
